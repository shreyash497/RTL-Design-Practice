  initial begin
    clk = 0;
    rst = 1;
    button = 0;
    #10 rst = 0;
    // posedge clk edges 5,15,25,35,45,55,65,75

    #10 button = 1; // 20ns

    #10 button = 0; // 60ns

    #20 $finish;
