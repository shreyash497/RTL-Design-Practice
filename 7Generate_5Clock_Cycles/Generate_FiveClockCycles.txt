module pulse_fsm(
    input clk,
    input rst,
    output reg pulse
);

reg [2:0] state;

parameter S0 = 3'd0,
          S1 = 3'd1,
          S2 = 3'd2,
          S3 = 3'd3,
          S4 = 3'd4;

always @(posedge clk or posedge rst)
begin
    if(rst)
    begin
        state <= S0;
        pulse <= 0;
    end
    else
    begin
        case(state)

        S0:
        begin
            state <= S1;
            pulse <= 0;
        end

        S1:
        begin
            state <= S2;
            pulse <= 0;
        end

        S2:
        begin
            state <= S3;
            pulse <= 0;
        end

        S3:
        begin
            state <= S4;
            pulse <= 0;
        end

        S4:
        begin
            state <= S0;
            pulse <= 1;
        end

        default:
        begin
            state <= S0;
            pulse <= 0;
        end

        endcase
    end
end

endmodule
