
//Detect Input HIGH for Two Consecutive Clock Cycles

module HighForTwoCycles( input clk ,rst ,x,
                        output reg [1:0] y );
  
  parameter s0=2'b00 , s1=2'b01, s2=2'b10 ;
  
  reg [1:0] state , next_state;
  
  always @(posedge clk )
    begin
      if (rst) 
        state<=s0;        
      else
        state<=next_state;
      
    end
  
  always @(*) begin
    case(state)
      
      s0: begin if(x)
        next_state=s1;
      else
        next_state=s0;
      end
      
      s1: begin if(x)
        next_state=s2;
      else
        next_state=s0;
      end
      
      s2: begin if(x)
        next_state=s2;
      else
        next_state=s0;
      end
      
      default:next_state=s0;
      
    endcase
    
   
      
      
  end
       always @(*) begin
    if(state == s2)
        y = 1'b1;
    else
        y = 1'b0;
end                 
endmodule
