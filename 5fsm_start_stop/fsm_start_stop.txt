//Design an FSM that turns ON an output after receiving a START //signal and turns it OFF after a STOP signal.
module fsm(
    input clk,
    input reset,
    input start,
    input stop,
    output reg out
);

reg state;

parameter OFF = 0,
          ON  = 1;

always @(posedge clk)
begin
    if(reset)
    begin
        state <= OFF;
        out <= 0;
    end

    else
    begin
        case(state)

            OFF:
            begin
                if(start)
                begin
                    state <= ON;
                    out <= 1;
                end
            end

            ON:
            begin
                if(stop)
                begin
                    state <= OFF;
                    out <= 0;
                end
            end

        endcase
    end
end

endmodule
