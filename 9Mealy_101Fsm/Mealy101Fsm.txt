module mealy_101_detector(
    input clk,
    input rst,
    input din,
    output reg y
);

    // State Declaration
    parameter S0 = 2'b00,
              S1 = 2'b01,
              S2 = 2'b10;

    reg [1:0] state, next_state;

    // State Transition
  
    always @(posedge clk or posedge rst) begin
        if(rst)
            state <= S0;
        else
            state <= next_state;
    end

    // Next State Logic + Output Logic
    always @(*) begin

        // Default values
        next_state = state;
        y = 1'b0;

        case(state)

            // S0 : No match
            S0: begin
                if(din == 1)
                    next_state = S1;
                else
                    next_state = S0;
            end

            // S1 : Detected 1
            S1: begin
                if(din == 0)
                    next_state = S2;
                else
                    next_state = S1;
            end

            // S2 : Detected 10
            S2: begin
                if(din == 1) begin
                    next_state = S1;
                    y = 1'b1;   // Sequence 101 detected
                end
                else begin
                    next_state = S0;
                    y = 1'b0;
                end
            end

            default: begin
                next_state = S0;
                y = 1'b0;
            end

        endcase
    end

endmodule
