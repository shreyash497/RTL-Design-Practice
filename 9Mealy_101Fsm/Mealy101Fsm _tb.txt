`timescale 1ns / 1ps

module tb_mealy_101_detector;

    reg clk;
    reg rst;
    reg din;
    wire y;

    // DUT Instantiation
    mealy_101_detector uut (
        .clk(clk),
        .rst(rst),
        .din(din),
        .y(y)
    );

    // Clock Generation
    always #5 clk = ~clk;

    initial begin

        // Waveform dump
        $dumpfile("mealy_101_detector.vcd");
        $dumpvars(0, tb_mealy_101_detector);

        // Initialize
        clk = 0;
        rst = 1;
        din = 0;

        // Apply Reset
        #10;
        rst = 0;

        // Input Sequence : 1 0 1 0 1

        #10 din = 1;
        #10 din = 0;
        #10 din = 1;
        #10 din = 0;
        #10 din = 1;

        #20;

        $finish;
    end

    // Monitor Signals
    initial begin
        $monitor("Time=%0t rst=%b din=%b y=%b",
                  $time, rst, din, y);
    end

endmodule
